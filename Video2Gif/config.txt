# ============================================================
#  Video2Gif 参数配置
#  用记事本打开改完保存即可，不用动 Video2Gif.cmd
#
#  规则，请遵守：
#   1. 参数行格式：键=值（半角等号，行首不要有空格）
#   2. 值只能是数字或 fast/normal/extra 这类简单单词
#      值里不要写引号、百分号、感叹号，也不要在行尾跟注释
#   3. 注释独立成行，以 # 开头
#   4. 只改值，别改键名，不认识的键会被忽略
#   5. 删掉某一行 = 该项恢复脚本内置默认值
#   6. 本文件必须保持 CRLF 行尾，记事本默认就是，别用会改行尾的编辑器
#
#  本文件用 UTF-8 或 GBK 或 ANSI 保存都能正常运行
#  当前按微信表情包场景预填：240px、单文件 500KB 以内
# ============================================================

# ---------- 基础 ----------

# WIDTH 输出宽度，像素，高度按比例自动算。240=表情包 480=普通 720=高清
WIDTH=240

# FPS 输出帧率。12 到 15 流畅够用，20 以上文件明显变大
FPS=15

# MAXDUR 只转前几秒，留空表示整段。长视频建议填 5，防止文件爆炸
MAXDUR=

# ---------- 体积，自动压到目标以下 ----------

# TARGETKB 目标体积上限，KB，纯数字。超过就自动降质量、帧率、宽度重编
# 微信表情包限制 500KB 就填 500；留空=不限制
# 降档顺序：质量 90/60/45/30，帧率 12/10，宽度 200/160，到底还超就提示截短 MAXDUR
# 注意：降档重试时 MOTIONQ 和 LOSSYQ 不参与，统一走总质量
TARGETKB=500

# ---------- 质量，gifski 管道模式生效 ----------

# Q 总质量 1-100，越高越清晰、体积越大。默认 90
# 设了 TARGETKB 就不用手动压这个：第一档尽量给 90，超了自动降
Q=90

# MOTIONQ 运动质量 1-100，控制快速运动画面的清晰度。留空=跟随 Q
MOTIONQ=

# LOSSYQ 有损压缩 1-100，输入画面是噪声照片时才需要调低。留空=关闭
LOSSYQ=

# SPEED 编码速度。fast=快一半画质略降  normal=默认  extra=慢一半画质略升
SPEED=normal

# REPEAT 循环次数。0=无限循环  -1=只播一次  3=循环 3 次
REPEAT=0

# ---------- 透明通道 ----------

# ALPHATH 透明分界阈值 0-255，仅 ffmpeg 兜底模式用，gifski 自己处理透明
ALPHATH=128

# ---------- 仅 ffmpeg 模式，没装 gifski 时用 ----------

# COLORQ 画质档位。1=文件更小但抖动粗糙  2=更细腻
COLORQ=2
